    function randomTitle() {

        const adjs = ["Crônicas do", "Lenda de", "O Último", "Batalha em", "Reino de", "Sombra do"];

    function adjustStat(statName, amount) {

        if (amount > 0 && pointsPool <= 0) return;

        if (amount < 0 && game[statName] <= 5) return;

        if (amount > 0) {

            game[statName] += 1;

            pointsPool -= 1;

        } else {

            game[statName] -= 1;

            pointsPool += 1;

        }

        document.getElementById('stat' + statName.charAt(0).toUpperCase() + statName.slice(1) + 'Val').innerText = game[statName];

        document.getElementById('pointsLeft').innerText = pointsPool;

    }

    function updateHours(type) {

        let p = parseInt(document.getElementById('rangePlot').value);

        let a = parseInt(document.getElementById('rangeArt').value);

        let r = parseInt(document.getElementById('rangeRest').value);

        let total = p + a + r;

            if(type === 'plot') document.getElementById('rangePlot').value = Math.max(0, 112 - a - r);

            if(type === 'art') document.getElementById('rangeArt').value = Math.max(0, 112 - p - r);

            if(type === 'rest') document.getElementById('rangeRest').value = Math.max(0, 112 - p - a);

            p = parseInt(document.getElementById('rangePlot').value);

            a = parseInt(document.getElementById('rangeArt').value);

            r = parseInt(document.getElementById('rangeRest').value);

        }

        document.getElementById('hrsPlot').innerText = p;

        document.getElementById('hrsRest').innerText = r;

    function finishWeek() {

        let p = parseInt(document.getElementById('rangePlot').value);

        let a = parseInt(document.getElementById('rangeArt').value);

        let r = parseInt(document.getElementById('rangeRest').value);

        let direction = document.getElementById('chapterDirection').value;

        if (direction === 'cliffhanger' || direction === 'action') {

            game.fatigue = Math.min(100, game.fatigue + 15);

        } else {

            game.fatigue = Math.max(0, game.fatigue - 10);

        }

        let workLoad = p + a;

        if (workLoad > 90) {

            game.health = Math.max(0, game.health - 8);

            game.inspiration = Math.max(0, game.inspiration - 5);

        } else {

            game.health = Math.min(100, game.health + 4);

            game.inspiration = Math.min(100, game.inspiration + r);

        }

        let quality = (p * 0.4) + (a * 0.4) + (game.inspiration * 0.2) - (game.fatigue * 0.2);

        let tocChange = Math.floor(Math.random() * 3) - 1;

        if (quality > 60) tocChange -= 1;

        else if (quality < 30) tocChange += 1;

        game.toc = Math.max(1, Math.min(15, game.toc + tocChange));

        game.chapter++;

        if (game.chapter % 9 === 0) {

            game.volumes++;

            let royalties = 300000 + (16 - game.toc) * 50000;

            game.money += royalties;

            logEvent(`Volume ${game.volumes} lançado! Lucro de ¥${royalties.toLocaleString()}.`);

        }

        if (game.toc >= 13) {

            game.dangerWeeks++;

            if (game.dangerWeeks >= 3) {

                alert("Sua obra foi cancelada pela revista devido às baixas colocações no TOC!");

                localStorage.removeItem('mangaka_tycoon_save');

                location.reload();

                return;

            } else {

                logEvent(`AVISO DO EDITOR: Seu mangá está nas posições de risco (${game.toc}º). Faltam ${3 - game.dangerWeeks} semanas para o cancelamento!`);

            }

        } else {

            game.dangerWeeks = 0;

        }

        updateUI();

        saveGame();

        logEvent(`Capítulo ${game.chapter - 1} publicado. Posição no TOC: #${game.toc}`);

    }
